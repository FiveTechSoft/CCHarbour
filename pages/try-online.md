# Try CCHarbour online

This page lets you quickly test the CCHarbour chat experience from your browser,
and check the latest Windows build generated by GitHub Actions.

!!! warning
    This web playground is for quick API testing only. The full CCHarbour
    terminal app (tools, shell, file editing, permissions) runs in `cc.exe`.
    Your API key is sent directly from your browser to the configured `Base URL`.
    Use only trusted endpoints. The key is kept in memory only for the current
    page session and is not encrypted.

## Latest CI build

- Workflow: [`build.yml`](https://github.com/FiveTechSoft/CCHarbour/actions/workflows/build.yml)
- Latest release binary: [cc.exe](https://github.com/FiveTechSoft/CCHarbour/releases/latest)

<script>
(() => {
  const owner = "FiveTechSoft";
  const repo = "CCHarbour";
  const workflowFile = "build.yml";
  const statusNode = document.getElementById("ci-status");
  const apiKey = document.getElementById("apiKey");
  const baseUrl = document.getElementById("baseUrl");
  const model = document.getElementById("model");
  const prompt = document.getElementById("prompt");
  const output = document.getElementById("output");
  const sendBtn = document.getElementById("sendBtn");

  baseUrl.value = localStorage.getItem("ccharbour.baseUrl") || baseUrl.value;
  model.value = localStorage.getItem("ccharbour.model") || model.value;

  const saveSettings = () => {
    localStorage.setItem("ccharbour.baseUrl", baseUrl.value.trim());
    localStorage.setItem("ccharbour.model", model.value.trim());
  };

  [baseUrl, model].forEach((el) => el.addEventListener("change", saveSettings));

  const loadBuildStatus = async () => {
    try {
      const url = `https://api.github.com/repos/${owner}/${repo}/actions/workflows/${workflowFile}/runs?per_page=1`;
      const res = await fetch(url, { headers: { "Accept": "application/vnd.github+json" } });
      if (!res.ok) throw new Error(`GitHub API ${res.status}`);
      const data = await res.json();
      const run = data.workflow_runs && data.workflow_runs[0];
      if (!run) {
        statusNode.textContent = "No workflow runs found yet.";
        return;
      }
      statusNode.innerHTML =
        `Latest run: <a href="${run.html_url}" target="_blank" rel="noopener">#${run.run_number}</a> ` +
        `(${run.event}) — status: <strong>${run.status}</strong>, conclusion: <strong>${run.conclusion || "pending"}</strong>`;
    } catch (err) {
      statusNode.textContent = `Could not load build status: ${err.message}`;
    }
  };

  const sendMessage = async () => {
    saveSettings();
    output.value = "";
    const key = apiKey.value.trim();
    const url = baseUrl.value.trim().replace(/\/+$/, "");
    const modelName = model.value.trim();
    const text = prompt.value.trim();

    if (!key || !url || !modelName || !text) {
      output.value = "Please fill API key, base URL, model and prompt.";
      return;
    }
    if (/\s/.test(key)) {
      output.value = "API key format looks invalid (must not contain spaces).";
      return;
    }
    if (!/^https:\/\//i.test(url)) {
      output.value = "Base URL must use HTTPS.";
      return;
    }

    sendBtn.disabled = true;
    sendBtn.textContent = "Sending...";

    try {
      const res = await fetch(`${url}/chat/completions`, {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${key}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          model: modelName,
          messages: [{ role: "user", content: text }],
          stream: false
        })
      });
      const data = await res.json();
      if (!res.ok) {
        output.value = data?.error?.message || "Request failed with an API error response.";
        return;
      }
      const content = data?.choices?.[0]?.message?.content;
      output.value = typeof content === "string" && content.length > 0
        ? content
        : "The API call succeeded, but no assistant message was returned.";
    } catch (err) {
      output.value = `Request failed: ${err.message}`;
    } finally {
      sendBtn.disabled = false;
      sendBtn.textContent = "Send";
    }
  };

  sendBtn.addEventListener("click", sendMessage);
  loadBuildStatus();
})();
</script>
